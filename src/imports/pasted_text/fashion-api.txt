/**
 * Figma Fashion Data API
 *
 * 목적:
 * - Google Spreadsheet를 패션 데이터 CMS처럼 사용
 * - Apps Script Web App으로 JSON API 제공
 * - Figma Plugin / Figma Make에서 fetch()로 데이터 조회
 *
 * 주요 함수:
 * - setupFashionSpreadsheet(): 최초 1회 실행
 * - doGet(e): Figma에서 데이터 조회
 * - doPost(e): 데이터 추가/수정/삭제
 */

/**
 * 최초 1회 실행
 *
 * 실행 결과:
 * - Google Spreadsheet 생성
 * - 헤더 생성
 * - 샘플 데이터 입력
 * - Script Properties에 SPREADSHEET_ID 저장
 */
function setupFashionSpreadsheet() {
  const ss = SpreadsheetApp.create(CONFIG.spreadsheetName);
  const sheet = ss.getActiveSheet();

  sheet.setName(CONFIG.sheetName);

  createHeader_(sheet);
  insertSampleRows_(sheet);
  applySheetFormatting_(sheet);
  applyDataValidation_(sheet);

  PropertiesService
    .getScriptProperties()
    .setProperty('SPREADSHEET_ID', ss.getId());

  Logger.log('Spreadsheet URL: ' + ss.getUrl());

  return {
    spreadsheetId: ss.getId(),
    spreadsheetUrl: ss.getUrl()
  };
}

/**
 * 쓰기 API용 토큰 설정
 *
 * 최초 1회 실행 예:
 * setApiToken()
 *
 * 실행 후 로그에 나온 토큰을 Figma나 관리자 도구에서 사용
 */
function setApiToken() {
  const token = Utilities.getUuid();

  PropertiesService
    .getScriptProperties()
    .setProperty('API_TOKEN', token);

  Logger.log('API_TOKEN: ' + token);

  return token;
}

/**
 * 현재 연결된 스프레드시트 URL 확인용
 */
function getSpreadsheetUrl() {
  const ss = getSpreadsheet_();
  Logger.log(ss.getUrl());
  return ss.getUrl();
}

/**
 * GET API
 *
 * 사용 예:
 * /exec
 * /exec?action=list
 * /exec?action=schema
 * /exec?action=health
 * /exec?id=LOOK_001
 * /exec?weather=sunny
 * /exec?gender=female
 * /exec?temperature_band=hot
 * /exec?weather=sunny&gender=female&temperature=28
 */
function doGet(e) {
  try {
    const params = e && e.parameter ? e.parameter : {};

    if (CONFIG.requireTokenForRead) {
      verifyToken_(params.token);
    }

    const action = params.action || 'list';

    if (action === 'health') {
      return jsonResponse_({
        success: true,
        message: 'Fashion API is running',
        timestamp: new Date()
      });
    }

    if (action === 'schema') {
      return jsonResponse_({
        success: true,
        sheetName: CONFIG.sheetName,
        headers: CONFIG.headers,
        options: {
          weather: CONFIG.weatherOptions,
          gender: CONFIG.genderOptions,
          temperature_band: CONFIG.temperatureBandOptions,
          fashion_category: CONFIG.fashionCategoryOptions
        }
      });
    }

    if (action === 'get') {
      const id = params.id;

      if (!id) {
        return errorResponse_('id 파라미터가 필요합니다.');
      }

      const item = getItemById_(id);

      if (!item) {
        return errorResponse_(`해당 id의 데이터를 찾을 수 없습니다: ${id}`);
      }

      return jsonResponse_({
        success: true,
        data: item
      });
    }

    const data = listItems_(params);

    return jsonResponse_({
      success: true,
      count: data.length,
      data
    });

  } catch (error) {
    return errorResponse_(error.message);
  }
}

/**
 * POST API
 *
 * 사용 예:
 *
 * {
 *   "token": "API_TOKEN",
 *   "action": "upsert",
 *   "item": {
 *     "id": "LOOK_004",
 *     "weather": "sunny",
 *     "temperature_avg_c": 27,
 *     "gender": "female",
 *     "fashion_category": "casual",
 *     "top": "white linen shirt",
 *     "bottom": "beige shorts",
 *     "shoes": "sandals",
 *     "description": "맑고 더운 날 여성 캐주얼 룩",
 *     "image_prompt": "Create a summer casual outfit for a woman..."
 *   }
 * }
 */
function doPost(e) {
  try {
    const body = parsePostBody_(e);

    if (CONFIG.requireTokenForWrite) {
      verifyToken_(body.token);
    }

    const action = body.action || 'upsert';

    if (action === 'add') {
      const created = addItem_(body.item);

      return jsonResponse_({
        success: true,
        action: 'add',
        data: created
      });
    }

    if (action === 'update') {
      const updated = updateItem_(body.item);

      return jsonResponse_({
        success: true,
        action: 'update',
        data: updated
      });
    }

    if (action === 'upsert') {
      const result = upsertItem_(body.item);

      return jsonResponse_({
        success: true,
        action: result.action,
        data: result.item
      });
    }

    if (action === 'delete') {
      const deleted = deleteItem_(body.id);

      return jsonResponse_({
        success: true,
        action: 'delete',
        data: deleted
      });
    }

    return errorResponse_(`지원하지 않는 action입니다: ${action}`);

  } catch (error) {
    return errorResponse_(error.message);
  }
}

/**
 * 데이터 목록 조회
 */
function listItems_(params) {
  const sheet = getSheet_();
  const values = sheet.getDataRange().getValues();

  if (values.length <= 1) {
    return [];
  }

  const headers = values[0];
  const rows = values.slice(1);

  let items = rows
    .filter(row => row[0])
    .map(row => rowToObject_(headers, row));

  items = items.filter(item => item.active === true || item.active === 'TRUE');

  if (params.id) {
    items = items.filter(item => String(item.id) === String(params.id));
  }

  if (params.weather) {
    items = items.filter(item => String(item.weather) === String(params.weather));
  }

  if (params.gender) {
    items = items.filter(item => String(item.gender) === String(params.gender));
  }

  if (params.fashion_category) {
    items = items.filter(item => String(item.fashion_category) === String(params.fashion_category));
  }

  if (params.temperature_band) {
    items = items.filter(item => String(item.temperature_band) === String(params.temperature_band));
  }

  /**
   * temperature 숫자로 조회하는 경우
   * 예: /exec?temperature=28
   *
   * temperature_band가 없어도 API에서 자동 계산해서 필터링 가능
   */
  if (params.temperature) {
    const temp = Number(params.temperature);
    const band = resolveTemperatureBand_(temp);

    items = items.filter(item => {
      if (item.temperature_band) {
        return String(item.temperature_band) === band;
      }

      const avg = Number(item.temperature_avg_c);

      if (Number.isNaN(avg)) {
        return false;
      }

      return resolveTemperatureBand_(avg) === band;
    });
  }

  if (params.limit) {
    const limit = Number(params.limit);

    if (!Number.isNaN(limit) && limit > 0) {
      items = items.slice(0, limit);
    }
  }

  return items;
}

/**
 * id 기준 단건 조회
 */
function getItemById_(id) {
  const items = listItems_({
    id
  });

  return items.length > 0 ? items[0] : null;
}

/**
 * 데이터 추가
 */
function addItem_(item) {
  if (!item) {
    throw new Error('item이 필요합니다.');
  }

  const sheet = getSheet_();
  const headers = getHeaders_(sheet);

  const normalized = normalizeItem_(item);

  if (!normalized.id) {
    normalized.id = generateId_();
  }

  const existingRowIndex = findRowIndexById_(sheet, normalized.id);

  if (existingRowIndex !== -1) {
    throw new Error(`이미 존재하는 id입니다: ${normalized.id}`);
  }

  normalized.updated_at = new Date();

  const row = objectToRow_(headers, normalized);
  sheet.appendRow(row);

  return normalized;
}

/**
 * 데이터 수정
 */
function updateItem_(item) {
  if (!item || !item.id) {
    throw new Error('수정할 item.id가 필요합니다.');
  }

  const sheet = getSheet_();
  const headers = getHeaders_(sheet);

  const rowIndex = findRowIndexById_(sheet, item.id);

  if (rowIndex === -1) {
    throw new Error(`수정할 데이터를 찾을 수 없습니다: ${item.id}`);
  }

  const currentRow = sheet
    .getRange(rowIndex, 1, 1, headers.length)
    .getValues()[0];

  const currentItem = rowToObject_(headers, currentRow);
  const mergedItem = Object.assign({}, currentItem, item);

  const normalized = normalizeItem_(mergedItem);
  normalized.updated_at = new Date();

  const newRow = objectToRow_(headers, normalized);

  sheet
    .getRange(rowIndex, 1, 1, headers.length)
    .setValues([newRow]);

  return normalized;
}

/**
 * 있으면 수정, 없으면 추가
 */
function upsertItem_(item) {
  if (!item) {
    throw new Error('item이 필요합니다.');
  }

  if (!item.id) {
    const created = addItem_(item);

    return {
      action: 'add',
      item: created
    };
  }

  const sheet = getSheet_();
  const rowIndex = findRowIndexById_(sheet, item.id);

  if (rowIndex === -1) {
    const created = addItem_(item);

    return {
      action: 'add',
      item: created
    };
  }

  const updated = updateItem_(item);

  return {
    action: 'update',
    item: updated
  };
}

/**
 * 데이터 삭제
 */
function deleteItem_(id) {
  if (!id) {
    throw new Error('삭제할 id가 필요합니다.');
  }

  const sheet = getSheet_();
  const headers = getHeaders_(sheet);

  const rowIndex = findRowIndexById_(sheet, id);

  if (rowIndex === -1) {
    throw new Error(`삭제할 데이터를 찾을 수 없습니다: ${id}`);
  }

  const row = sheet
    .getRange(rowIndex, 1, 1, headers.length)
    .getValues()[0];

  const item = rowToObject_(headers, row);

  sheet.deleteRow(rowIndex);

  return item;
}

 * 헤더 생성

  sheet
    .getRange(1, 1, 1, CONFIG.headers.length)
    .setValues([CONFIG.headers]);

/**
 * 시트 서식 적용
 */
function applySheetFormatting_(sheet) {
  sheet.setFrozenRows(1);

  sheet
    .getRange(1, 1, 1, CONFIG.headers.length)
    .setFontWeight('bold')
    .setBackground('#eeeeee');

  sheet.autoResizeColumns(1, CONFIG.headers.length);

  const activeColumnIndex = CONFIG.headers.indexOf('active') + 1;

  if (activeColumnIndex > 0) {
    sheet
      .getRange(2, activeColumnIndex, Math.max(sheet.getMaxRows() - 1, 1), 1)
      .insertCheckboxes();
  }
}

/**
 * 드롭다운 검증 적용
 */
function applyDataValidation_(sheet) {
  applyDropdown_(sheet, 'weather', CONFIG.weatherOptions);
  applyDropdown_(sheet, 'gender', CONFIG.genderOptions);
  applyDropdown_(sheet, 'temperature_band', CONFIG.temperatureBandOptions);
  applyDropdown_(sheet, 'fashion_category', CONFIG.fashionCategoryOptions);
}

/**
 * 특정 컬럼에 드롭다운 적용
 */
function applyDropdown_(sheet, headerName, options) {
  const columnIndex = CONFIG.headers.indexOf(headerName) + 1;

  if (columnIndex <= 0) {
    return;
  }

  const rule = SpreadsheetApp
    .newDataValidation()
    .requireValueInList(options, true)
    .setAllowInvalid(false)
    .build();

  sheet
    .getRange(2, columnIndex, Math.max(sheet.getMaxRows() - 1, 1), 1)
    .setDataValidation(rule);
}

/**
 * 스프레드시트 가져오기
 */
function getSpreadsheet_() {
  const spreadsheetId = PropertiesService
    .getScriptProperties()
    .getProperty('SPREADSHEET_ID');

  if (!spreadsheetId) {
    throw new Error('SPREADSHEET_ID가 없습니다. setupFashionSpreadsheet()를 먼저 실행하세요.');
  }

  return SpreadsheetApp.openById(spreadsheetId);
}

/**
 * 시트 가져오기
 */
function getSheet_() {
  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName(CONFIG.sheetName);

  if (!sheet) {
    throw new Error(`Sheet not found: ${CONFIG.sheetName}`);
  }

  return sheet;
}

/**
 * 헤더 가져오기
 */
function getHeaders_(sheet) {
  return sheet
    .getRange(1, 1, 1, sheet.getLastColumn())
    .getValues()[0];
}

/**
 * row 배열을 object로 변환
 */
function rowToObject_(headers, row) {
  const item = {};

  headers.forEach((header, index) => {
    item[header] = row[index];
  });

  return item;
}

/**
 * object를 row 배열로 변환
 */
function objectToRow_(headers, item) {
  return headers.map(header => {
    if (item[header] === undefined || item[header] === null) {
      return '';
    }

    return item[header];
  });
}

/**
 * 입력 item 정규화
 */
function normalizeItem_(item) {
  const normalized = Object.assign({}, item);

  normalized.weather = normalizeString_(normalized.weather);
  normalized.gender = normalizeString_(normalized.gender);
  normalized.temperature_band = normalizeString_(normalized.temperature_band);
  normalized.fashion_category = normalizeString_(normalized.fashion_category);

  normalized.temperature_min_c = normalizeNumber_(normalized.temperature_min_c);
  normalized.temperature_max_c = normalizeNumber_(normalized.temperature_max_c);
  normalized.temperature_avg_c = normalizeNumber_(normalized.temperature_avg_c);
  normalized.temperature_feels_like_c = normalizeNumber_(normalized.temperature_feels_like_c);

  if (!normalized.temperature_band && normalized.temperature_avg_c !== '') {
    normalized.temperature_band = resolveTemperatureBand_(Number(normalized.temperature_avg_c));
  }

  if (normalized.active === undefined || normalized.active === null || normalized.active === '') {
    normalized.active = true;
  }

  validateItem_(normalized);

  return normalized;
}

/**
 * 데이터 검증
 */
function validateItem_(item) {
  if (!item.weather) {
    throw new Error('weather는 필수입니다.');
  }

  if (!CONFIG.weatherOptions.includes(item.weather)) {
    throw new Error(`지원하지 않는 weather입니다: ${item.weather}`);
  }

  if (!item.gender) {
    throw new Error('gender는 필수입니다.');
  }

  if (!CONFIG.genderOptions.includes(item.gender)) {
    throw new Error(`지원하지 않는 gender입니다: ${item.gender}`);
  }

  if (item.temperature_band && !CONFIG.temperatureBandOptions.includes(item.temperature_band)) {
    throw new Error(`지원하지 않는 temperature_band입니다: ${item.temperature_band}`);
  }

  if (item.fashion_category && !CONFIG.fashionCategoryOptions.includes(item.fashion_category)) {
    throw new Error(`지원하지 않는 fashion_category입니다: ${item.fashion_category}`);
  }
}

/**
 * 기온을 temperature_band로 변환
 */
function resolveTemperatureBand_(temperatureC) {
  if (temperatureC <= 0) {
    return 'freezing';
  }

  if (temperatureC <= 9) {
    return 'cold';
  }

  if (temperatureC <= 16) {
    return 'cool';
  }

  if (temperatureC <= 22) {
    return 'mild';
  }

  if (temperatureC <= 27) {
    return 'warm';
  }

  return 'hot';
}

/**
 * id로 row index 찾기
 *
 * 주의:
 * Apps Script의 row index는 1부터 시작
 * 헤더가 1행이므로 데이터는 2행부터 시작
 */
function findRowIndexById_(sheet, id) {
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) {
    return -1;
  }

  const idColumnIndex = CONFIG.headers.indexOf('id') + 1;

  const ids = sheet
    .getRange(2, idColumnIndex, lastRow - 1, 1)
    .getValues()
    .flat();

  const foundIndex = ids.findIndex(value => String(value) === String(id));

  if (foundIndex === -1) {
    return -1;
  }

  return foundIndex + 2;
}

/**
 * POST body 파싱
 */
function parsePostBody_(e) {
  if (!e || !e.postData || !e.postData.contents) {
    throw new Error('POST body가 비어 있습니다.');
  }

  return JSON.parse(e.postData.contents);
}

/**
 * API 토큰 검증
 */
function verifyToken_(requestToken) {
  const savedToken = PropertiesService
    .getScriptProperties()
    .getProperty('API_TOKEN');

  if (!savedToken) {
    throw new Error('API_TOKEN이 설정되어 있지 않습니다. setApiToken()을 먼저 실행하세요.');
  }

  if (!requestToken || requestToken !== savedToken) {
    throw new Error('유효하지 않은 API token입니다.');
  }
}

/**
 * 문자열 정규화
 */
function normalizeString_(value) {
  if (value === undefined || value === null) {
    return '';
  }

  return String(value).trim().toLowerCase();
}

/**
 * 숫자 정규화
 */
function normalizeNumber_(value) {
  if (value === undefined || value === null || value === '') {
    return '';
  }

  const numberValue = Number(value);

  if (Number.isNaN(numberValue)) {
    return '';
  }

  return numberValue;
}

/**
 * id 자동 생성
 */
function generateId_() {
  const uuid = Utilities.getUuid().split('-')[0].toUpperCase();
  return `LOOK_${uuid}`;
}

/**
 * JSON 응답
 */
function jsonResponse_(payload) {
  return ContentService
    .createTextOutput(JSON.stringify(payload, null, 2))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * 에러 응답
 */
function errorResponse_(message) {
  return jsonResponse_({
    success: false,
    error: message
  });
}
